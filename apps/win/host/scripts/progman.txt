# Program Manager as the shell: run with "./winhost scripts/progman.txt PROGMAN"
# after "rm -f HOSTWIN/PROGMAN.INI" (the first start fills the Main group).
wait 100
snap pm_start.ppm
# File > New Program Group
keydown ALT
key F
keyup ALT
key G
wait 50
type Games
key RETURN
wait 50
# File > New Program Item, into the new (active) group
keydown ALT
key F
keyup ALT
key I
wait 50
type Second Demo
key TAB
type DEMO.EXE
key RETURN
wait 50
snap pm_groups.ppm
# Window > Tile
keydown SHIFT
key F4
keyup SHIFT
wait 50
snap pm_tile.ppm
# click into Main (left half): it becomes active; Window menu lists both
click 100 150
wait 50
keydown ALT
key W
keyup ALT
wait 50
snap pm_winmenu.ppm
key ESC
key ESC
# maximize Main with its caption button
dblclick 150 74
wait 50
snap pm_max.ppm
# Ctrl+F6: Games, maximized in turn
keydown CTRL
key F6
keyup CTRL
wait 50
snap pm_next.ppm
# restore, cascade
dblclick 150 74
wait 50
keydown SHIFT
key F5
keyup SHIFT
wait 50
snap pm_cascade.ppm
# Ctrl+F4 on a group minimizes it
keydown CTRL
key F4
keyup CTRL
wait 50
snap pm_min.ppm
# double click Demo in Main
dblclick 64 120
wait 200
snap pm_demo.ppm
keydown ALT
key F4
keyup ALT
wait 100
# File > Run with a DOS program
keydown ALT
key F
keyup ALT
key R
wait 50
type NC
key RETURN
wait 50
snap pm_runerr.ppm
key RETURN
wait 50
# closing the shell ends Windows
keydown ALT
key F4
keyup ALT
wait 50
snap pm_exit.ppm
key RETURN
wait 100
exit
